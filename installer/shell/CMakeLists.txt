cmake_minimum_required(VERSION 3.20)
project(skipit_installer LANGUAGES CXX RC)
option(SKIPIT_PREVIEW "Build a preview that cannot install anything" OFF)
if(NOT DEFINED APP_VERSION)
  message(FATAL_ERROR "APP_VERSION is required")
endif()
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" NUMERIC_VERSION "${APP_VERSION}")
if(NOT NUMERIC_VERSION)
  message(FATAL_ERROR "APP_VERSION must start with three numeric components")
endif()
string(REPLACE "." "," RESOURCE_VERSION "${NUMERIC_VERSION}")
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../windows/runner/resources/app_icon.ico" APP_ICON)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../license.txt" LICENSE_FILE)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../brand-mark.png" MARK_FILE)
if(SKIPIT_PREVIEW)
  set(EXECUTION_LEVEL "asInvoker")
  set(PAYLOAD_RESOURCE "")
else()
  if(NOT EXISTS "${INSTALLER_PAYLOAD}")
    message(FATAL_ERROR "INSTALLER_PAYLOAD is required")
  endif()
  file(TO_CMAKE_PATH "${INSTALLER_PAYLOAD}" INSTALLER_PAYLOAD)
  set(EXECUTION_LEVEL "requireAdministrator")
  set(PAYLOAD_RESOURCE "102 RCDATA \"${INSTALLER_PAYLOAD}\"")
endif()
configure_file(shell.rc.in shell.rc @ONLY)
configure_file(shell.manifest.in shell.manifest @ONLY)
add_executable(SkipItSetup WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/shell.rc" "${CMAKE_CURRENT_BINARY_DIR}/shell.manifest")
target_compile_features(SkipItSetup PRIVATE cxx_std_17)
target_compile_definitions(SkipItSetup PRIVATE UNICODE _UNICODE NOMINMAX "APP_VERSION=L\"${APP_VERSION}\"" "$<$<BOOL:${SKIPIT_PREVIEW}>:SKIPIT_PREVIEW>")
target_compile_options(SkipItSetup PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_options(SkipItSetup PRIVATE /MANIFESTUAC:NO)
set_property(TARGET SkipItSetup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(SkipItSetup PRIVATE comctl32 dwmapi gdiplus shell32 ole32 advapi32 bcrypt uxtheme shlwapi)
if(SKIPIT_PREVIEW)
  enable_testing()
  add_executable(SkipItSetupUiTests WIN32 ui_tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/shell.rc" "${CMAKE_CURRENT_BINARY_DIR}/shell.manifest")
  target_compile_features(SkipItSetupUiTests PRIVATE cxx_std_17)
  target_compile_definitions(SkipItSetupUiTests PRIVATE UNICODE _UNICODE NOMINMAX SKIPIT_PREVIEW "APP_VERSION=L\"${APP_VERSION}\"")
  target_compile_options(SkipItSetupUiTests PRIVATE /W4 /WX /utf-8 /EHsc)
  target_link_options(SkipItSetupUiTests PRIVATE /MANIFESTUAC:NO)
  set_property(TARGET SkipItSetupUiTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(SkipItSetupUiTests PRIVATE comctl32 dwmapi gdiplus shell32 ole32 advapi32 bcrypt uxtheme shlwapi)
  add_test(NAME installer_ui COMMAND SkipItSetupUiTests)
  set_tests_properties(installer_ui PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
